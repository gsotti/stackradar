  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "logpilot.fullname" . }})

{{- else if contains "LoadBalancer" .Values.service.type }}
Get the application URL by running these commands:
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "logpilot.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "logpilot.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
Get the application URL by running these commands:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "logpilot.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

📋 Quick Start:
1. Register a new user at the web interface
2. Create a system to get an API token
3. Send logs to: POST /api/ingest/{api_token}

📤 Log Ingestion Example:
  curl -X POST "http://YOUR_URL/api/ingest/YOUR_API_TOKEN" \
    -H "Content-Type: application/json" \
    -d '{"logs": [{"level": "INFO", "message": "Hello from my app", "source": "my-service"}]}'

🔧 Important Notes:
- Change the JWT_SECRET and ADMIN_API_KEY in production!
- The cleanup job runs every hour by default
- Data is stored in a PersistentVolumeClaim

📊 Kubernetes Metrics:
To send K8s metrics, POST to: /api/k8s/metrics/{api_token}

For more information, visit the documentation.
